cmake_minimum_required(VERSION 3.22.1)
project(aicluster LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_Declare(llama URL https://github.com/ggml-org/llama.cpp/archive/refs/tags/b5046.zip URL_HASH SHA256=628626865cdc521bcfcbd3f11dd1ca8a229e13899132847138814e88d060ebf6)
FetchContent_MakeAvailable(llama)
add_library(aicluster SHARED benchmark.cpp)
target_compile_features(aicluster PRIVATE cxx_std_17)
target_compile_options(aicluster PRIVATE -O3)
target_link_libraries(aicluster PRIVATE llama android log c++_shared)
